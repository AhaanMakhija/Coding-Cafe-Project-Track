{# Renders one WTForms field with its label, an optional hint and any errors. extra_class lets a page tag a field, e.g. "only-file", so CSS can hide it. #} {% macro field(f, hint=None, extra_class="") %}
{% if f.errors %} {{ f(aria_invalid="true", aria_describedby=f.id ~ "-error") }} {{ f.errors | join(" ") }} {% else %} {{ f() }} {% endif %} {% if hint %}{{ hint }}{% endif %}
{% endmacro %}
